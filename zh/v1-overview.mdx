API v1 的基础路径为 `https://api.neural4d.com/openapi/v1`。每个请求都需要携带 Neural4D 提供的 API 密钥。

## 调用流程

<Steps>
  <Step title="查询可用模型">
    查询模型目录，并根据返回的能力构造请求。

    <CodeGroup>
      ```bash cURL
      curl 'https://api.neural4d.com/openapi/v1/models' \
        --header 'Authorization: Bearer ${YOUR_API_KEY}'
      ```

      ```python Python
      import os
      import requests

      response = requests.get(
          "https://api.neural4d.com/openapi/v1/models",
          headers={"Authorization": f"Bearer {os.environ['YOUR_API_KEY']}"},
      )
      print(response.json())
      ```

      ```javascript Node.js
      const response = await fetch("https://api.neural4d.com/openapi/v1/models", {
        headers: {
          Authorization: `Bearer ${process.env.YOUR_API_KEY}`,
        },
      });

      console.log(await response.json());
      ```

      ```go Go
      package main

      import (
          "fmt"
          "io"
          "net/http"
          "os"
      )

      func main() {
          request, _ := http.NewRequest("GET", "https://api.neural4d.com/openapi/v1/models", nil)
          request.Header.Set("Authorization", "Bearer "+os.Getenv("YOUR_API_KEY"))

          response, _ := http.DefaultClient.Do(request)
          defer response.Body.Close()
          body, _ := io.ReadAll(response.Body)
          fmt.Println(string(body))
      }
      ```
    </CodeGroup>
  </Step>

  <Step title="提交生成任务">
    创建类接口返回 `202 Accepted`，每个请求输出对应一个任务对象。

    <CodeGroup>
      ```bash cURL
      curl 'https://api.neural4d.com/openapi/v1/images/generations' \
        --header 'Authorization: Bearer ${YOUR_API_KEY}' \
        --header 'Content-Type: application/json' \
        --data '{
          "model": "openai/gpt-image-2",
          "prompt": "白色摄影棚背景中的陶瓷茶壶",
          "mode": "text-to-image",
          "n": 1,
          "resolution": "1K",
          "aspect_ratio": "1:1"
        }'
      ```

      ```python Python
      import os
      import requests

      response = requests.post(
          "https://api.neural4d.com/openapi/v1/images/generations",
          headers={"Authorization": f"Bearer {os.environ['YOUR_API_KEY']}"},
          json={
              "model": "openai/gpt-image-2",
              "prompt": "白色摄影棚背景中的陶瓷茶壶",
              "mode": "text-to-image",
              "n": 1,
              "resolution": "1K",
              "aspect_ratio": "1:1",
          },
      )
      print(response.json())
      ```

      ```javascript Node.js
      const response = await fetch("https://api.neural4d.com/openapi/v1/images/generations", {
        method: "POST",
        headers: {
          Authorization: `Bearer ${process.env.YOUR_API_KEY}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          model: "openai/gpt-image-2",
          prompt: "白色摄影棚背景中的陶瓷茶壶",
          mode: "text-to-image",
          n: 1,
          resolution: "1K",
          aspect_ratio: "1:1",
        }),
      });

      console.log(await response.json());
      ```

      ```go Go
      package main

      import (
          "bytes"
          "fmt"
          "io"
          "net/http"
          "os"
      )

      func main() {
          body := []byte(`{"model":"openai/gpt-image-2","prompt":"白色摄影棚背景中的陶瓷茶壶","mode":"text-to-image","n":1,"resolution":"1K","aspect_ratio":"1:1"}`)
          request, _ := http.NewRequest("POST", "https://api.neural4d.com/openapi/v1/images/generations", bytes.NewReader(body))
          request.Header.Set("Authorization", "Bearer "+os.Getenv("YOUR_API_KEY"))
          request.Header.Set("Content-Type", "application/json")

          response, _ := http.DefaultClient.Do(request)
          defer response.Body.Close()
          result, _ := io.ReadAll(response.Body)
          fmt.Println(string(result))
      }
      ```
    </CodeGroup>
  </Step>

  <Step title="轮询任务状态">
    保存返回任务的 `id`，然后调用 `GET /openapi/v1/tasks/{id}`，直到状态变为 `succeeded` 或 `failed`。
  </Step>

  <Step title="获取结果文件">
    任务成功后调用 `GET /openapi/v1/files/{id}`。返回的 URL 带签名且有时效性，应将 `id` 作为永久标识保存。
  </Step>
</Steps>

认证、请求追踪和限流响应头请参阅[通用请求头](/zh/common-headers)。
