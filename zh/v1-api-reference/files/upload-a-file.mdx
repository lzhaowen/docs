## 授权

<ParamField header="Authorization" type="string" required default={'Bearer ${YOUR_API_KEY}'}>
  使用 Bearer 方式发送的 Neural4D API 密钥。
</ParamField>

## 请求体 <span className="api-content-type">multipart/form-data</span>

<ParamField body="file" type="string&lt;binary&gt;" required>
  需要上传的文件。图片支持 JPG、JPEG、PNG、GIF、WEBP、BMP、HEIC 和 HEIF；视频支持 MP4、MOV、WEBM、M4V、OGG 和 OGV；音频支持 MP3 和 WAV。

  默认单文件上限：图片 20 MiB、视频 100 MiB、音频 15 MiB。超过上传上限返回 `400 file_too_large`。
</ParamField>

生成请求会根据图片用途再次校验大小，以下上限同时适用于上传返回的 `file_id` 和公网 HTTPS `url`：

| 图片用途 | 每张图片上限 |
| --- | --- |
| 生图参考图片 | 10 MiB |
| 生视频首帧、尾帧及参考图片 | 10 MiB |
| 单图、批量图生 3D | 20 MiB |
| 多视图生 3D | 10 MiB |

```java Java · 上传文件
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.UUID;

class Main {
  public static void main(String[] args) throws Exception {
    Path file = Path.of("/path/to/file.png");
    String boundary = "----Neural4DUpload" + UUID.randomUUID();
    String partHeader = "--" + boundary + "\r\n"
        + "Content-Disposition: form-data; name=\"file\"; filename=\""
        + file.getFileName() + "\"\r\n"
        + "Content-Type: image/png\r\n\r\n";

    HttpRequest.BodyPublisher body = HttpRequest.BodyPublishers.ofByteArrays(List.of(
        partHeader.getBytes(StandardCharsets.UTF_8),
        Files.readAllBytes(file),
        ("\r\n--" + boundary + "--\r\n").getBytes(StandardCharsets.UTF_8)
    ));
    HttpRequest request = HttpRequest.newBuilder()
        .uri(URI.create("https://api.neural4d.com/openapi/v1/files"))
        .header("Authorization", "Bearer <token>")
        .header("Content-Type", "multipart/form-data; boundary=" + boundary)
        .POST(body)
        .build();
    HttpResponse<String> response = HttpClient.newHttpClient().send(
        request,
        HttpResponse.BodyHandlers.ofString()
    );
    System.out.println(response.body());
  }
}
```

```go Go · 上传文件
package main

import (
	"bytes"
	"fmt"
	"io"
	"log"
	"mime"
	"mime/multipart"
	"net/http"
	"net/textproto"
	"os"
	"path/filepath"
)

func main() {
	filename := "/path/to/file.png"
	data, err := os.ReadFile(filename)
	if err != nil {
		log.Fatal(err)
	}

	var payload bytes.Buffer
	writer := multipart.NewWriter(&payload)
	header := make(textproto.MIMEHeader)
	header.Set("Content-Disposition", mime.FormatMediaType(
		"form-data",
		map[string]string{
			"name":     "file",
			"filename": filepath.Base(filename),
		},
	))
	contentType := mime.TypeByExtension(filepath.Ext(filename))
	if contentType == "" {
		contentType = http.DetectContentType(data)
	}
	header.Set("Content-Type", contentType)
	part, err := writer.CreatePart(header)
	if err != nil {
		log.Fatal(err)
	}
	if _, err := part.Write(data); err != nil {
		log.Fatal(err)
	}
	if err := writer.Close(); err != nil {
		log.Fatal(err)
	}

	req, err := http.NewRequest(
		"POST",
		"https://api.neural4d.com/openapi/v1/files",
		&payload,
	)
	if err != nil {
		log.Fatal(err)
	}
	req.Header.Set("Authorization", "Bearer <token>")
	req.Header.Set("Content-Type", writer.FormDataContentType())
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(string(body))
}
```

```ruby Ruby · 上传文件
require "net/http"

uri = URI("https://api.neural4d.com/openapi/v1/files")
request = Net::HTTP::Post.new(uri)
request["Authorization"] = "Bearer <token>"

File.open("/path/to/file.png", "rb") do |file|
  request.set_form(
    [["file", file, {
      filename: File.basename(file.path),
      content_type: "image/png"
    }]],
    "multipart/form-data"
  )

  response = Net::HTTP.start(
    uri.hostname,
    uri.port,
    use_ssl: uri.scheme == "https"
  ) do |http|
    http.request(request)
  end
  puts response.body
end
```
