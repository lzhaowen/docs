The API v1 base URL is `https://api.neural4d.com/openapi/v1`. This guide uses text-to-3D to show the core workflow.

## 1. Set your API key

Use an account with an active paid subscription to get your key from the [API key page](https://www.neural4d.com/account?api).

```bash Configure your API key
export YOUR_API_KEY='your-api-key'
```

## 2. Create a 3D generation task

Submit a prompt to generate a 3D model with PBR materials. Set both `with_texture: true` and `with_pbr: true`. Send your key in the `Authorization: Bearer` header with every request.

<div data-api-example="request">
<CodeGroup>
```bash cURL
curl --request POST \
  --url 'https://api.neural4d.com/openapi/v1/3dmodels/generations' \
  --header "Authorization: Bearer ${YOUR_API_KEY}" \
  --header 'Content-Type: application/json' \
  --data '{
  "model": "dreamtech/neural4d-2.5",
  "prompt": "A ceramic teapot on a white studio background",
  "with_texture": true,
  "with_pbr": true
}'
```

```java Java
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

class Main {
  public static void main(String[] args) throws Exception {
    String apiKey = System.getenv("YOUR_API_KEY");

    String payload = String.join("\n",
        "{",
        "  \"model\": \"dreamtech/neural4d-2.5\",",
        "  \"prompt\": \"A ceramic teapot on a white studio background\",",
        "  \"with_texture\": true,",
        "  \"with_pbr\": true",
        "}"
    );

    HttpRequest request = HttpRequest.newBuilder()
        .uri(URI.create("https://api.neural4d.com/openapi/v1/3dmodels/generations"))
        .header("Authorization", "Bearer " + apiKey)
        .header("Content-Type", "application/json")
        .POST(HttpRequest.BodyPublishers.ofString(payload))
        .build();
    HttpResponse<String> response = HttpClient.newHttpClient().send(
        request,
        HttpResponse.BodyHandlers.ofString()
    );
    System.out.println(response.body());
  }
}
```

```go Go
package main

import (
    "fmt"
    "io"
    "log"
    "net/http"
    "os"
    "strings"
)

func main() {
    if err := run(); err != nil {
        log.Fatal(err)
    }
}

func run() error {
    payload := strings.NewReader(`{
      "model": "dreamtech/neural4d-2.5",
      "prompt": "A ceramic teapot on a white studio background",
      "with_texture": true,
      "with_pbr": true
    }`)
    request, err := http.NewRequest(
        "POST",
        "https://api.neural4d.com/openapi/v1/3dmodels/generations",
        payload,
    )
    if err != nil {
        return err
    }
    request.Header.Set("Authorization", "Bearer "+os.Getenv("YOUR_API_KEY"))
    request.Header.Set("Content-Type", "application/json")

    response, err := http.DefaultClient.Do(request)
    if err != nil {
        return err
    }
    defer response.Body.Close()
    result, err := io.ReadAll(response.Body)
    if err != nil {
        return err
    }
    fmt.Println(string(result))
    return nil
}
```

## 3. Retrieve and download the result

Replace the sample `id` with the value returned when you created the task, then query every few seconds. `queued` and `processing` mean the task is still running. Stop at `succeeded` or `failed`; check `error` if it failed.

<div data-api-example="request">
<CodeGroup>
```bash cURL
curl --request GET \
  --url 'https://api.neural4d.com/openapi/v1/tasks/task-info?id=normal-3d-c50fe63e-699d-4d61-93f5-2099ab159d6d' \
  --header "Authorization: Bearer ${YOUR_API_KEY}"
```

```python Python
import os
import requests

api_key = os.environ["YOUR_API_KEY"]

    "https://api.neural4d.com/openapi/v1/tasks/task-info?id=normal-3d-c50fe63e-699d-4d61-93f5-2099ab159d6d",
    headers={
        "Authorization": f"Bearer {api_key}"
    }
)
print(response.json())
```
