## Authorizations

<ParamField header="Authorization" type="string" required default={'Bearer ${YOUR_API_KEY}'}>
  Use the bearer token provided through the Neural4D website.
</ParamField>

## Body <span className="api-content-type">multipart/form-data</span>

<ParamField body="prompt" type="string" required>
  Text prompt describing the desired generated video.
  </ParamField>
<ParamField body="images" type="string&lt;binary&gt;[]">
  Optional reference images used to guide generation.

Allowed array length: `0 - 6` elements.
</ParamField>
<ParamField body="aspectRatio" type="enum&lt;string&gt;" default="16:9">
  Target aspect ratio for the generated video.

Available options: `16:9`, `9:16`, `1:1`, `4:3`, `3:4`.
</ParamField>
<ParamField body="jobNum" type="integer" default={1}>
  Number of generation jobs to submit.

Required range: `1 <= x <= 4`.
</ParamField>
<ParamField body="duration" type="integer" default={5}>
  Target video duration in seconds.

Required range: `4 <= x <= 15`.
</ParamField>
<ParamField body="resolution" type="enum&lt;string&gt;" default="720p">
  Target output resolution.

Available options: `480p`, `720p`, `1080p`.
</ParamField>

## Response

<Tabs sync={false}>
  <Tab title="200">
    Video generation job accepted
        
    <ResponseField name="success" type="boolean" required>
      Whether the generation request was accepted.
    </ResponseField>

    <ResponseField name="data" type="object" required>
      Queue and task details for the accepted generation request.

      <Expandable title="child attributes">
        <ResponseField name="pointsDeducted" pre={["data."]} type="integer" required>
          Points deducted for the accepted generation request.
        </ResponseField>

        <ResponseField name="queueStatus" pre={["data."]} type="string" required>
          Current queue state for the submitted job.
        </ResponseField>

        <ResponseField name="jobNum" pre={["data."]} type="integer" required>
          Number of accepted generation jobs.
        </ResponseField>

        <ResponseField name="uuids" pre={["data."]} type="string[]" required>
          UUIDs assigned to each accepted generation job.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Tab>
</Tabs>
