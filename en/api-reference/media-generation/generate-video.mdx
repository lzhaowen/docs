## Authorizations

<ParamField header="Authorization" type="string" required default={'Bearer ${YOUR_API_KEY}'}>
  Use the bearer token provided through the Neural4D website.
</ParamField>

## Body <span className="api-content-type">multipart/form-data</span>

<ParamField body="prompt" type="string" required>
  Text prompt describing the desired generated video.
  </ParamField>
<ParamField body="images" type="string&lt;binary&gt;[]">
  Optional reference images used to guide generation.

Allowed array length: `0 - 6` elements.
</ParamField>
<ParamField body="aspectRatio" type="enum&lt;string&gt;" default="16:9">
  Target aspect ratio for the generated video.

Available options: `16:9`, `9:16`, `1:1`, `4:3`, `3:4`.
</ParamField>
<ParamField body="jobNum" type="integer" default={1}>
  Number of generation jobs to submit.

Required range: `1 <= x <= 4`.
</ParamField>
<ParamField body="duration" type="integer" default={5}>
  Target video duration in seconds.

Required range: `4 <= x <= 15`.
</ParamField>
<ParamField body="resolution" type="enum&lt;string&gt;" default="720p">
  Target output resolution.

Available options: `480p`, `720p`, `1080p`.
</ParamField>

```go Go
package main

import (
	"bytes"
	"fmt"
	"io"
	"log"
	"mime"
	"mime/multipart"
	"net/http"
	"net/textproto"
	"os"
	"path/filepath"
)

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	var payload bytes.Buffer
	writer := multipart.NewWriter(&payload)

	fields := []struct {
		name  string
		value string
	}{
		{"prompt", "a flower bouquet"},
		{"duration", "14"},
		{"aspectRatio", "9:16"},
		{"resolution", "480p"},
		{"jobNum", "3"},
	}
	for _, field := range fields {
		if err := writer.WriteField(field.name, field.value); err != nil {
			return err
		}
	}

	files := []struct {
		fieldName string
		filePath  string
	}{
		{"images", "/path/to/file.png"},
	}
	for _, file := range files {
		if err := addFile(writer, file.fieldName, file.filePath); err != nil {
			return err
		}
	}
	if err := writer.Close(); err != nil {
		return err
	}

	req, err := http.NewRequest(
		"POST",
		"https://alb.neural4d.com:3000/api/createNormalVideo",
		&payload,
	)
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer <token>")
	req.Header.Set("Content-Type", writer.FormDataContentType())

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	bodyBytes, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}
	fmt.Println(string(bodyBytes))
	return nil
}

func addFile(writer *multipart.Writer, fieldName, filePath string) error {
	fileBytes, err := os.ReadFile(filePath)
	if err != nil {
		return err
	}
	header := make(textproto.MIMEHeader)
	header.Set("Content-Disposition", mime.FormatMediaType("form-data", map[string]string{
		"name":     fieldName,
		"filename": filepath.Base(filePath),
	}))
	contentType := mime.TypeByExtension(filepath.Ext(filePath))
	if contentType == "" {
		contentType = http.DetectContentType(fileBytes)
	}
	header.Set("Content-Type", contentType)
	part, err := writer.CreatePart(header)
	if err != nil {
		return err
	}
	_, err = part.Write(fileBytes)
	return err
}
```

    <ResponseField name="success" type="boolean" required>

    </ResponseField>

    <ResponseField name="data" type="object" required>

        <ResponseField name="queueStatus" pre={["data."]} type="string" required>
          Current queue state for the submitted job.
        </ResponseField>

        <ResponseField name="jobNum" pre={["data."]} type="integer" required>
          Number of accepted generation jobs.
        </ResponseField>

        <ResponseField name="uuids" pre={["data."]} type="string[]" required>
          UUIDs assigned to each accepted generation job.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Tab>
