## Body <span className="api-content-type">multipart/form-data</span>

<ParamField body="images" type="string&lt;binary&gt;[]" required>
  Input image files used to generate chibi-style models.

Allowed array length: `1 - unbounded` elements.
</ParamField>
<ParamField body="styleType" type="string" required>
  Configured prompt tag used by the cute-model workflow.

Example: `"Pixel-Style"`.
</ParamField>

```go Go
package main

import (
	"bytes"
	"fmt"
	"io"
	"log"
	"mime"
	"mime/multipart"
	"net/http"
	"net/textproto"
	"os"
	"path/filepath"
)

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	var payload bytes.Buffer
	writer := multipart.NewWriter(&payload)

	fields := []struct {
		name  string
		value string
	}{
		{"styleType", "Pixel-Style"},
	}
	for _, field := range fields {
		if err := writer.WriteField(field.name, field.value); err != nil {
			return err
		}
	}

	files := []struct {
		fieldName string
		filePath  string
	}{
		{"images", "/path/to/file.png"},
	}
	for _, file := range files {
		if err := addFile(writer, file.fieldName, file.filePath); err != nil {
			return err
		}
	}
	if err := writer.Close(); err != nil {
		return err
	}

	req, err := http.NewRequest(
		"POST",
		"https://alb.neural4d.com:3000/api/createCuteModelsFromImages",
		&payload,
	)
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer <token>")
	req.Header.Set("Content-Type", writer.FormDataContentType())

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	bodyBytes, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}
	fmt.Println(string(bodyBytes))
	return nil
}

func addFile(writer *multipart.Writer, fieldName, filePath string) error {
	fileBytes, err := os.ReadFile(filePath)
	if err != nil {
		return err
	}
	header := make(textproto.MIMEHeader)
	header.Set("Content-Disposition", mime.FormatMediaType("form-data", map[string]string{
		"name":     fieldName,
		"filename": filepath.Base(filePath),
	}))
	contentType := mime.TypeByExtension(filepath.Ext(filePath))
	if contentType == "" {
		contentType = http.DetectContentType(fileBytes)
	}
	header.Set("Content-Type", contentType)
	part, err := writer.CreatePart(header)
	if err != nil {
		return err
	}
	_, err = part.Write(fileBytes)
	return err
}
```
