```go Go
package main

import (
	"bytes"
	"fmt"
	"io"
	"log"
	"mime"
	"mime/multipart"
	"net/http"
	"net/textproto"
	"os"
	"path/filepath"
)

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}

func run() error {
	var payload bytes.Buffer
	writer := multipart.NewWriter(&payload)

	files := []struct {
		fieldName string
		filePath  string
	}{
		{"image", "/path/to/file.png"},
	}
	for _, file := range files {
		if err := addFile(writer, file.fieldName, file.filePath); err != nil {
			return err
		}
	}
	if err := writer.Close(); err != nil {
		return err
	}

	req, err := http.NewRequest(
		"POST",
		"https://alb.neural4d.com:3000/api/checkHumanImage",
		&payload,
	)
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer <token>")
	req.Header.Set("Content-Type", writer.FormDataContentType())

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	bodyBytes, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}
	fmt.Println(string(bodyBytes))
	return nil
}

func addFile(writer *multipart.Writer, fieldName, filePath string) error {
	fileBytes, err := os.ReadFile(filePath)
	if err != nil {
		return err
	}
	header := make(textproto.MIMEHeader)
	header.Set("Content-Disposition", mime.FormatMediaType("form-data", map[string]string{
		"name":     fieldName,
		"filename": filepath.Base(filePath),
	}))
	contentType := mime.TypeByExtension(filepath.Ext(filePath))
	if contentType == "" {
		contentType = http.DetectContentType(fileBytes)
	}
	header.Set("Content-Type", contentType)
	part, err := writer.CreatePart(header)
	if err != nil {
		return err
	}
	_, err = part.Write(fileBytes)
	return err
}
```

## Response

<Tabs sync={false}>
  <Tab title="200">
    Detection result
        
    <ResponseField name="statusCode" type="enum&lt;integer&gt;" required>
      Detection status. `0` means the result is valid and `-1` means the result is invalid.

    Available options: `-1`, `0`.
    </ResponseField>



    <ResponseField name="result" type="boolean" required>
      Whether a human portrait was detected in the uploaded image.
    </ResponseField>



    <ResponseField name="message" type="string" required>
      Human-readable portrait detection result.
    </ResponseField>
  </Tab>
</Tabs>
