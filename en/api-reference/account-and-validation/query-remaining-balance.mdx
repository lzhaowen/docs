## Response

<Tabs sync={false}>
  <Tab title="200">
    Remaining points information
        
    <ResponseField name="statusType" type="enum&lt;integer&gt;" required>
      Query status. `0` means the result is valid and `-1` means the result is invalid.

    Available options: `-1`, `0`.
    </ResponseField>

    <ResponseField name="data" type="object" required>
      Current point balances grouped by point type.

      <Expandable title="child attributes">
        <ResponseField name="total_points" pre={["data."]} type="integer" required>
          Total points currently available to the account.
        </ResponseField>

        <ResponseField name="monthly_bonus_points" pre={["data."]} type="integer" required>
          Available points granted by the monthly plan.
        </ResponseField>

        <ResponseField name="permanent_points" pre={["data."]} type="integer" required>
          Available permanent points without a monthly expiration.
        </ResponseField>

        <ResponseField name="year_points" pre={["data."]} type="integer" required>
          Available points granted by the annual plan.
        </ResponseField>

        <ResponseField name="expiration_day" pre={["data."]} type="string | null" required>
          Expiration day for time-limited points when present.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="message" type="string" required>
      Human-readable point balance query result.
    </ResponseField>
  </Tab>
</Tabs>
